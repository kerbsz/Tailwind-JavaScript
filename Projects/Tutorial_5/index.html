class PcItems{
        constructor(name,brand,price,discount){
            this.name = name;
            this.brand = brand;
            this.price = price;
            this.discount = discount;
        }

        displayProduct(){
            console.log(`Product name: ${this.name}`);
            console.log(`Product brand: ${this.brand}`);
            console.log(`Product price: ${this.price}`);
            console.log(`Product discount: ${this.discount * 100}%`);
            console.log(`Total Price: ${this.calculateDiscount()}`);
        }

        calculateDiscount(discount){
            return this.price - (this.price * this.discount);
        }
}

//---------------------Static-------------------//
class Mathutil{
    static pi = 3.14;

    static getDiameter(radius){ 
        return radius * 2;
    }
    
    static getCircumference(radius){
        return 2 * this.pi * radius;
    }
    
    static getArea(radius){
        return this.pi * radius * radius;
    }
}

//-----------------Inheritance------------------//
class Animal{
    alive = true;

    eat() {console.log(`This ${this.name} is eating!`);}
    sleeping() {console.log(`This ${this.name} is sleeping!`);}
}
class Dog extends Animal{
    name = "milky";

    canWalktoLand() {console.log(`I can walk to land`)}
}

class bird extends Animal{
    name = "thirdy";

    canWalktoLand() {console.log(`I can walk to land`);}
    canFly() {console.log(`I can fly`);}
}
const dog = new Dog();
const parrot = new bird();

dog.eat();
dog.sleeping();
dog.canWalktoLand();

parrot.canFly();
